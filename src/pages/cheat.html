A 題骨架

// 學號 姓名
`timescale 1ns/1ps
module exam1_A (
    input  logic               clk, rst,
    input  logic signed [7:0]  A, B,
    input  logic        [1:0]  ctrl,
    output logic signed [15:0] out
);
    logic signed [15:0] result;
    always_comb begin
        case (ctrl)
            2'b00:   result = A * B;
            2'b01:   result = {A & B, A ^ B};
            2'b11:   result = (A + B) <<< 2;
            default: result = (A < B) ? -16'sd1 : 16'sd1;
        endcase
    end
    always_ff @(posedge clk) begin
        if (rst) out <= '0;
        else     out <= result;
    end
endmodule

B 題骨架（規則計數器）

module exam1_B (input logic clk, rst, output logic [15:0] out);
    localparam logic [16:0] MAX = 17'd65176;      // 題目給的上限
    logic        up;
    logic [15:0] i, prev;
    logic [16:0] nxt;                 // 多 1 bit 看溢位 / 負數

    always_comb begin
        if (up) nxt = /* 上數規則 */ out + i;
        else    nxt = (out < i) ? '0 : out - i;   // 先比較，避免無號數變超大
    end

    always_ff @(posedge clk) begin
        if (rst) begin out <= '0; prev <= '0; i <= 16'd1; up <= 1'b1; end
        else begin
            prev <= out;
            out  <= nxt[15:0];
            if (up && nxt >= MAX)      begin up <= 1'b0; i <= 16'd1; end
            else if (!up && nxt == 0)  begin up <= 1'b1; i <= 16'd1; end
            else                        i <= i + 16'd1;
        end
    end
endmodule

C 題骨架

clock_divider、debounce、one_pulse、SevenSegment 考試會給（通常已經在範本檔案裡），下面只示範怎麼接。

module exam1_C (
    input  logic clk, rst, en, set, up, down,
    input  logic [15:0] sw,
    output logic [3:0]  DIGIT,
    output logic [6:0]  DISPLAY,
    output logic [15:0] led
);
    // 1) 按鈕：debounce（慢時脈取樣）-> one_pulse（100 MHz，1 個 clk 寬）
    logic clk_db;
    clock_divider #(.n(16)) u_cd_db (.clk(clk), .clk_div(clk_db));
    logic en_d, set_d, up_d, down_d, en_p, set_p, up_p, down_p;
    debounce  u_db0 (.clk(clk_db), .pb(en),   .pb_debounced(en_d));
    debounce  u_db1 (.clk(clk_db), .pb(set),  .pb_debounced(set_d));
    debounce  u_db2 (.clk(clk_db), .pb(up),   .pb_debounced(up_d));
    debounce  u_db3 (.clk(clk_db), .pb(down), .pb_debounced(down_d));
    one_pulse u_op0 (.clk(clk), .pb_in(en_d),   .pb_out(en_p));
    one_pulse u_op1 (.clk(clk), .pb_in(set_d),  .pb_out(set_p));
    one_pulse u_op2 (.clk(clk), .pb_in(up_d),   .pb_out(up_p));
    one_pulse u_op3 (.clk(clk), .pb_in(down_d), .pb_out(down_p));

    // 2) tick：clock_divider + one_pulse(~clk_div)，每 2^n 週期一個 clk 寬的脈衝
    logic cd25, t25;
    clock_divider #(.n(25)) u_cd25 (.clk(clk), .clk_div(cd25));
    one_pulse     u_t25    (.clk(clk), .pb_in(~cd25), .pb_out(t25));

    // 3) 狀態
    typedef enum logic [1:0] {S_RESET, S_START, S_PAUSE, S_FINISH} state_t;
    state_t state;

    // 4) 資料暫存器 + 狀態轉移：一個 always_ff，全部用 clk
    logic [3:0] pos, score;
    always_ff @(posedge clk) begin
        if (rst) begin
            state <= S_RESET; pos <= 4'd15; score <= '0;
        end else case (state)
            S_RESET: if (en_p) state <= S_START;
            S_START: begin
                if (en_p) state <= S_PAUSE;
                if (t25)  pos <= pos - 4'd1;
            end
            S_PAUSE: if (en_p) state <= S_START;
            default: ;
        endcase
    end

    // 5) 輸出：LED 遮罩、nums 交給 SevenSegment（nums[15:12] 最左）
    logic [15:0] nums;
    assign led  = 16'b1 << pos;
    assign nums = {4'd0, 4'd0, 4'd0, score};
    SevenSegment u_seg (.display(DISPLAY), .digit(DIGIT), .nums(nums), .rst(rst), .clk(clk));
endmodule

一定要記得的規則

有號數

整個式子都 signed 才做有號運算。常數用 16'sd3；part-select 和串接是無號；需要時 $signed()。右移負數用 >>>。

寬度

運算前先擴到整個式子最寬的寬度。加法多 1 bit、乘法寬度相加。寬給窄直接砍高位。

always

always_comb 用 =，先給預設值、case 加 default。always_ff 用 <=。一個訊號只在一個 always 裡指定。

reset

同步：@(posedge clk)；非同步：@(posedge clk or posedge rst)。考試 A/B/C 都寫同步。

時間

224 ≈ 0.17 s、225 ≈ 0.34 s、226 ≈ 0.67 s、227 ≈ 1.34 s。剛好 1 秒 = 100,000,000 週期（27 bits）。

不用 / %

x % 2^k = x[k-1:0]、x / 2^k = x >> k、x*3 = (x<<1)+x、奇偶看 x[0]。

LED 方向

led[15] 在最左。往左 = index + 1。4-bit 位置加減自動環繞。

七段

DIGIT、DISPLAY 都是 0 = 亮。DIGIT 1110 = 最右位。DISPLAY = {g,f,e,d,c,b,a}。
