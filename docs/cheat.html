A 題骨架

// 學號 姓名
`timescale 1ns/1ps
module exam1_A (
    input  logic               clk, rst,
    input  logic signed [7:0]  A, B,
    input  logic        [1:0]  ctrl,
    output logic signed [15:0] out
);
    logic signed [15:0] result;
    always_comb begin
        case (ctrl)
            2'b00:   result = A * B;
            2'b01:   result = {A & B, A ^ B};
            2'b11:   result = (A + B) <<< 2;
            default: result = (A < B) ? -16'sd1 : 16'sd1;
        endcase
    end
    always_ff @(posedge clk) begin
        if (rst) out <= '0;
        else     out <= result;
    end
endmodule

B 題骨架（規則計數器）

module exam1_B (input logic clk, rst, output logic [15:0] out);
    localparam logic [16:0] MAX = 17'd65176;      // 題目給的上限
    logic        up;
    logic [15:0] i, prev;
    logic [16:0] nxt;                 // 多 1 bit 看溢位 / 負數

    always_comb begin
        if (up) nxt = /* 上數規則 */ out + i;
        else    nxt = (out < i) ? '0 : out - i;   // 先比較，避免無號數變超大
    end

    always_ff @(posedge clk) begin
        if (rst) begin out <= '0; prev <= '0; i <= 16'd1; up <= 1'b1; end
        else begin
            prev <= out;
            out  <= nxt[15:0];
            if (up && nxt >= MAX)      begin up <= 1'b0; i <= 16'd1; end
            else if (!up && nxt == 0)  begin up <= 1'b1; i <= 16'd1; end
            else                        i <= i + 16'd1;
        end
    end
endmodule

C 題骨架

module exam1_C (
    input  logic clk, rst, en, set, up, down,
    input  logic [15:0] sw,
    output logic [3:0]  DIGIT,
    output logic [6:0]  DISPLAY,
    output logic [15:0] led
);
    // 1) 按鈕 -> 1 週期脈衝
    logic en_p, set_p, up_p, down_p;
    btn_pulse u0 (.clk, .rst, .btn(en),  .pulse(en_p));
    btn_pulse u1 (.clk, .rst, .btn(set), .pulse(set_p));
    btn_pulse u2 (.clk, .rst, .btn(up),  .pulse(up_p));
    btn_pulse u3 (.clk, .rst, .btn(down),.pulse(down_p));

    // 2) 狀態
    typedef enum logic [1:0] {S_RESET, S_START, S_PAUSE, S_FINISH} state_t;
    state_t state;

    // 3) tick：同一個計數器
    logic [25:0] cnt;
    logic t24, t25, t26;
    always_ff @(posedge clk) cnt <= rst ? '0 : cnt + 1'b1;
    assign t24 = (cnt[23:0] == '1);
    assign t25 = (cnt[24:0] == '1);
    assign t26 = (cnt == '1);

    // 4) 資料暫存器 + 狀態轉移：一個 always_ff
    logic [3:0] pos;  logic [3:0] score;
    always_ff @(posedge clk) begin
        if (rst) begin
            state <= S_RESET; pos <= 4'd15; score <= '0;
        end else case (state)
            S_RESET: if (en_p) state <= S_START;
            S_START: begin
                if (en_p) state <= S_PAUSE;
                if (t25)  pos <= pos - 4'd1;
            end
            S_PAUSE: if (en_p) state <= S_START;
            default: ;
        endcase
    end

    // 5) 輸出
    assign led = 16'b1 << pos;
    seven_seg u_seg (.clk, .c3(5'd0), .c2(5'd0), .c1(5'd0), .c0({1'b0, score}), .DIGIT, .DISPLAY);
endmodule

// ===================================================================
//  Reusable helper modules for FPGA problems (Basys3, 100 MHz clock)
//  Paste these at the bottom of your exam file: the exam requires
//  ONE file that contains every module you use.
// ===================================================================

// -------------------------------------------------------------------
// btn_pulse: debounce + one-pulse in ONE clock domain (100 MHz).
//   * The button is sampled once every 2^DB clock cycles
//     (DB = 16 -> every 0.65 ms), shifted into a 4-bit register.
//   * "stable" = the last 4 samples were all 1.
//   * pulse   = rising edge of "stable", exactly ONE 100 MHz cycle wide.
// -------------------------------------------------------------------
module btn_pulse #(parameter int DB = 16) (
    input  logic clk,
    input  logic rst,
    input  logic btn,
    output logic pulse
);
    logic [DB-1:0] div;
    logic [3:0]    samples;
    logic          stable, stable_d;

    assign stable = &samples;

    always_ff @(posedge clk) begin
        if (rst) begin
            div <= '0; samples <= '0; stable_d <= 1'b0;
        end else begin
            div <= div + 1'b1;
            if (div == '0) samples <= {samples[2:0], btn};
            stable_d <= stable;
        end
    end

    assign pulse = stable & ~stable_d;
endmodule

// -------------------------------------------------------------------
// seven_seg: 4-digit multiplexed display driver.
//   * c3 is the LEFTMOST digit, c0 the RIGHTMOST.
//   * each cN is a 5-bit character code (see the localparams below).
//   * DIGIT and DISPLAY are ACTIVE-LOW on Basys3.
//   * DISPLAY = {g, f, e, d, c, b, a}
// -------------------------------------------------------------------
module seven_seg (
    input  logic       clk,
    input  logic [4:0] c3, c2, c1, c0,
    output logic [3:0] DIGIT,
    output logic [6:0] DISPLAY
);
    // character codes: 0..9 are the digits themselves
    localparam logic [4:0] CH_A = 5'd10, CH_E = 5'd14, CH_P = 5'd16,
                           CH_R = 5'd17, CH_U = 5'd18, CH_DASH = 5'd19,
                           CH_BLANK = 5'd31;
    // 'S' looks like 5 and 'I' looks like 1 -> just use 5 and 1

    logic [16:0] refresh = '0;     // 2^15 cycles per digit (~0.33 ms); '= 0' sets the FPGA power-up value
    logic [4:0]  ch;

    always_ff @(posedge clk) refresh <= refresh + 1'b1;

    always_comb begin
        case (refresh[16:15])
            2'd0: begin DIGIT = 4'b1110; ch = c0; end
            2'd1: begin DIGIT = 4'b1101; ch = c1; end
            2'd2: begin DIGIT = 4'b1011; ch = c2; end
            default: begin DIGIT = 4'b0111; ch = c3; end
        endcase
    end

    always_comb begin
        case (ch)
            5'd0: DISPLAY = 7'b1000000;
            5'd1: DISPLAY = 7'b1111001;
            5'd2: DISPLAY = 7'b0100100;
            5'd3: DISPLAY = 7'b0110000;
            5'd4: DISPLAY = 7'b0011001;
            5'd5: DISPLAY = 7'b0010010;
            5'd6: DISPLAY = 7'b0000010;
            5'd7: DISPLAY = 7'b1111000;
            5'd8: DISPLAY = 7'b0000000;
            5'd9: DISPLAY = 7'b0010000;
            CH_A: DISPLAY = 7'b0001000;
            CH_E: DISPLAY = 7'b0000110;
            CH_P: DISPLAY = 7'b0001100;
            CH_R: DISPLAY = 7'b0101111;   // lowercase r
            CH_U: DISPLAY = 7'b1000001;
            CH_DASH: DISPLAY = 7'b0111111;
            default: DISPLAY = 7'b1111111; // blank
        endcase
    end
endmodule

一定要記得的規則

有號數

整個式子都 signed 才做有號運算。常數用 16'sd3；part-select 和串接是無號；需要時 $signed()。右移負數用 >>>。

寬度

運算前先擴到整個式子最寬的寬度。加法多 1 bit、乘法寬度相加。寬給窄直接砍高位。

always

always_comb 用 =，先給預設值、case 加 default。always_ff 用 <=。一個訊號只在一個 always 裡指定。

reset

同步：@(posedge clk)；非同步：@(posedge clk or posedge rst)。考試 A/B/C 都寫同步。

時間

224 ≈ 0.17 s、225 ≈ 0.34 s、226 ≈ 0.67 s、227 ≈ 1.34 s。剛好 1 秒 = 100,000,000 週期（27 bits）。

不用 / %

x % 2^k = x[k-1:0]、x / 2^k = x >> k、x*3 = (x<<1)+x、奇偶看 x[0]。

LED 方向

led[15] 在最左。往左 = index + 1。4-bit 位置加減自動環繞。

七段

DIGIT、DISPLAY 都是 0 = 亮。DIGIT 1110 = 最右位。DISPLAY = {g,f,e,d,c,b,a}。
