Lab 1：Gate-level 設計

學什麼

用 and/or/not/nand 等 primitive 組電路；模組實例化（依名稱接）。

考試會用到

Practice 1：用 2 選 1 MUX 做 function selector

C = 0 輸出 A&B，C = 1 輸出 A|B。先用閘做 MUX（y = a·c' + b·c），再實例化。

解答：practice_1.sv（testbench 8/8 Correct）
// Lab 1 Practice 1 — function selector with a gate-level 2-to-1 MUX
`timescale 1ns/100ps
module practice_1 (
    input  wire A,
    input  wire B,
    input  wire C,
    output wire out
);
    wire and_ab, or_ab;
    and g_and (and_ab, A, B);          // primitive: output first, then inputs
    or  g_or  (or_ab,  A, B);
    mux u_mux (.a(and_ab), .b(or_ab), .c(C), .y(out));   // C=0 -> AND, C=1 -> OR
endmodule

module mux (
    input  wire a,
    input  wire b,
    input  wire c,
    output wire y
);
    wire c_n, t0, t1;
    not g0 (c_n, c);
    and g1 (t0, a, c_n);    // a & ~c
    and g2 (t1, b, c);      // b &  c
    or  g3 (y, t0, t1);
endmodule

Practice 2：Gated D latch

G = 1 時 P 跟著 D（透明）；G = 0 時保持。用兩個 NAND 交叉回授做 SR latch，前面再用兩個 NAND 控制 set/reset。

解答：practice_2.sv（testbench All tests passed）
// Lab 1 Practice 2 — gated D latch, gate level only (NAND version)
`timescale 1ns/100ps
module practice_2 (
    input  logic G,
    input  logic D,
    output logic P,
    output logic Pn
);
    logic D_n, s_n, r_n;
    not  g0 (D_n, D);
    nand g1 (s_n, D,   G);   // set   (active low) when G=1 and D=1
    nand g2 (r_n, D_n, G);   // reset (active low) when G=1 and D=0
    nand g3 (P,  s_n, Pn);   // cross-coupled NAND = SR latch
    nand g4 (Pn, r_n, P);
endmodule
Latch vs Flip-flopLatch 是「準位觸發」：G = 1 的整段時間都會跟著 D 變。Flip-flop 是「邊緣觸發」：只在 clk 上升那一瞬間取樣。考試的同步電路都用 flip-flop（always_ff @(posedge clk)）；在 always_comb 意外生出的 latch 是 bug。

Practice 1：8-bit 有號飽和加法器

A + B 超過 127 就輸出 127，低於 -128 就輸出 -128，否則正常輸出。輸出要暫存、同步 reset。

關鍵：用 9 bits 的中間變數算和，就不會溢位，再判斷範圍。

解答：lab2_1_practice.sv（PASS 32/32）
// Lab 2 Practice 1 — registered 8-bit signed SATURATING adder
`timescale 1ns/1ps
module lab2_1_practice (
    input  logic              clk,
    input  logic              rst,     // synchronous
    input  logic signed [7:0] A,
    input  logic signed [7:0] B,
    output logic signed [7:0] out
);
    logic signed [8:0] sum;            // one extra bit: never overflows
    logic signed [7:0] sat;

    always_comb begin
        sum = A + B;                   // 9-bit context -> both operands sign-extended
        if      (sum >  9'sd127)  sat = 8'sd127;
        else if (sum < -9'sd128)  sat = -8'sd128;
        else                      sat = sum[7:0];
    end

    always_ff @(posedge clk) begin
        if (rst) out <= '0;
        else     out <= sat;
    end
endmodule

Practice 2：可載入的倒數計數器

load = 1 時 out 載入 N；之後每拍減 1，減到 0 的那一拍 done = 1（只有一拍），之後停在 0。

解答：lab2_2_practice.sv（PASS 32/32）
// Lab 2 Practice 2 — loadable down counter with a one-cycle "done" flag
`timescale 1ns/1ps
module lab2_2_practice (
    input  logic       clk,
    input  logic       rst,    // synchronous
    input  logic       load,
    input  logic [7:0] N,
    output logic [7:0] out,
    output logic       done
);
    always_ff @(posedge clk) begin
        if (rst) begin
            out  <= 8'd0;
            done <= 1'b0;
        end else if (load) begin
            out  <= N;
            done <= 1'b0;
        end else if (out != 8'd0) begin
            out  <= out - 8'd1;
            done <= (out == 8'd1);      // we are about to reach 0
        end else begin
            done <= 1'b0;
        end
    end
endmodule
易錯done 是暫存器，要在「out 從 1 變 0」的同一個邊緣設成 1，所以條件是 out == 1（舊值），不是 out == 0。

解答：lab3_practice.sv
// Lab 3 Practice — LED controller FSM (INITIAL -> LOAD -> PLAY -> FINAL -> INITIAL)
`timescale 1ns/1ps
module lab3_practice (
    input  logic        clk,
    input  logic        rst,        // asynchronous, active-high (SW15)
    input  logic        slow,       // SW0
    input  logic        fast,       // SW1
    input  logic        end_light,  // SW2
    output logic [15:0] led
);
    typedef enum logic [1:0] {INITIAL, LOAD, PLAY, FINAL} state_t;
    state_t state, next_state;

    // ---- timers: one-cycle "tick" pulses (exact time, not powers of two) ----
    logic tick_q, tick_h;                     // 0.25 s and 0.5 s
    timer #(.COUNT(25_000_000)) t_q (.clk, .rst, .enable(state == LOAD),  .tick(tick_q));
    timer #(.COUNT(50_000_000)) t_h (.clk, .rst, .enable(state == FINAL), .tick(tick_h));

    // ---- clock dividers for the flashing in PLAY (2^28 and 2^27) ----
    logic div_slow, div_fast;
    clock_divider #(.n(28)) d_s (.clk, .rst, .enable(state == PLAY), .clk_div(div_slow));
    clock_divider #(.n(27)) d_f (.clk, .rst, .enable(state == PLAY), .clk_div(div_fast));

    logic [4:0] load_cnt;   // how many LEDs are on in LOAD (0..16)
    logic [2:0] half_cnt;   // number of finished 0.5 s periods in FINAL (0..5)

    // ---- state register (async reset) ----
    always_ff @(posedge clk or posedge rst) begin
        if (rst) state <= INITIAL;
        else     state <= next_state;
    end

    // ---- next-state logic ----
    always_comb begin
        next_state = state;
        case (state)
            INITIAL: next_state = LOAD;
            LOAD:    if (load_cnt == 5'd16)          next_state = PLAY;
            PLAY:    if (end_light)                  next_state = FINAL;
            FINAL:   if (tick_h && half_cnt == 3'd5) next_state = INITIAL;
        endcase
    end

    // ---- datapath counters ----
    always_ff @(posedge clk or posedge rst) begin
        if (rst) begin
            load_cnt <= '0; half_cnt <= '0;
        end else begin
            if (state != LOAD)                    load_cnt <= '0;
            else if (tick_q && load_cnt != 5'd16) load_cnt <= load_cnt + 5'd1;

            if (state != FINAL) half_cnt <= '0;
            else if (tick_h)    half_cnt <= half_cnt + 3'd1;
        end
    end

    // ---- outputs (Moore: depend on state + datapath only) ----
    always_comb begin
        case (state)
            INITIAL: led = 16'hFFFF;
            LOAD:    led = ~(16'hFFFF << load_cnt);                     // load_cnt LEDs on, from LD0
            PLAY: begin
                led = 16'h0000;
                if (slow) led = led | (div_slow ? 16'h5555 : 16'h0000);  // even LEDs
                if (fast) led = led | (div_fast ? 16'hAAAA : 16'h0000);  // odd LEDs
            end
            default: led = half_cnt[0] ? 16'h0000 : 16'hFFFF;           // FINAL: on/off x3
        endcase
    end
endmodule

// ---- modules given in the template ----
module timer #(parameter integer COUNT = 100_000_000) (
    input logic clk, input logic rst, input logic enable, output logic tick
);
    logic [31:0] num;
    always_ff @(posedge clk or posedge rst) begin
        if (rst)                   num <= 32'd0;
        else if (!enable)          num <= 32'd0;
        else if (num == COUNT - 1) num <= 32'd0;
        else                       num <= num + 1'b1;
    end
    assign tick = enable && (num == COUNT - 1);
endmodule

module clock_divider #(parameter n = 10) (
    input logic clk, input logic rst, input logic enable, output logic clk_div
);
    logic [n-1:0] num;
    always_ff @(posedge clk or posedge rst) begin
        if (rst)         num <= '0;
        else if (enable) num <= num + 1'b1;
        else             num <= '0;
    end
    assign clk_div = num[n-1];
endmodule
重點
  • 除頻器（2 的次方）做不出剛好 0.25 秒；要剛好的時間就用 timer 計數到 N-1 產生 1 拍的 tick。
  • timer 的 enable 接「目前狀態 == 某狀態」，離開狀態自動歸零，下次進來重新計時。
  • ~(16'hFFFF << n) 產生最低 n 個 bit 為 1 的遮罩。

Lab 3 advanced 摘要

  • 2 位數加法器：SET1 → SET2 → RESULT 的 FSM，用 BTNC 切換；七段顯示 BCD 數字，正在設定的兩位每 0.5 秒閃爍；結果 0–198 要拆成百十個位。
  • 跑馬燈：在四位七段顯示上循環捲動 "nthuEECS2026"，BTNL/BTNR 改方向，BTNC 暫停，SW0 改速度（1 秒或 0.5 秒）。核心是用一個「起始位置」指標，四位數分別顯示字串的第 k、k+1、k+2、k+3 個字。

Lab 4：按鈕、LFSR、方向遊戲

INIT 狀態按 BTNR 讓 LFSR 前進（選起始值），按 BTNU 開始。PLAY 狀態的目標方向是 lfsr[1:0]，按對了 LFSR 前進一步，按錯進 FINAL，LED 全亮 3 秒後回 INIT。

stateDiagram-v2
    direction LR
    INIT --> INIT: btnR / lfsr 前進
    INIT --> PLAY: btnU
    PLAY --> PLAY: 按對 / lfsr 前進
    PLAY --> FINAL: 按錯
    FINAL --> INIT: 3 秒後 / lfsr = 0001
解答：lab4_practice.sv
// Lab 4 Practice — direction game: 4-bit LFSR + one FSM
`timescale 1ns/1ps
module lab4_basic (
    input  logic        clk,
    input  logic        rst,     // asynchronous, active-high (BTNC)
    input  logic        btnL,
    input  logic        btnR,
    input  logic        btnD,
    input  logic        btnU,
    output logic [15:0] LED
);
    typedef enum logic [1:0] {INIT, PLAY, FINAL} state_t;
    state_t state;

    // ---- every button -> one clean pulse ----
    logic pL, pR, pD, pU;
    btn_pulse u_l (.clk, .rst, .btn(btnL), .pulse(pL));
    btn_pulse u_r (.clk, .rst, .btn(btnR), .pulse(pR));
    btn_pulse u_d (.clk, .rst, .btn(btnD), .pulse(pD));
    btn_pulse u_u (.clk, .rst, .btn(btnU), .pulse(pU));

    // ---- 4-bit LFSR: shift left, feedback = lfsr[3] ^ lfsr[0]  (0001 -> 0011 -> 0111 ...) ----
    logic [3:0] lfsr, lfsr_next;
    assign lfsr_next = {lfsr[2:0], lfsr[3] ^ lfsr[0]};

    // ---- which button was pressed (Left 00, Right 01, Down 10, Up 11) ----
    logic       any_p;
    logic [1:0] code;
    assign any_p = pL | pR | pD | pU;
    always_comb begin
        if      (pL) code = 2'b00;
        else if (pR) code = 2'b01;
        else if (pD) code = 2'b10;
        else         code = 2'b11;
    end

    // ---- 3-second timer in FINAL ----
    parameter ONE_SEC = 27'd100_000_000;
    logic [26:0] sec_cnt;
    logic [1:0]  secs;

    always_ff @(posedge clk or posedge rst) begin
        if (rst) begin
            state <= INIT; lfsr <= 4'b0001; sec_cnt <= '0; secs <= '0;
        end else begin
            case (state)
                INIT: begin
                    if (pR)      lfsr  <= lfsr_next;            // pick the starting state
                    else if (pU) state <= PLAY;
                end
                PLAY: begin
                    if (any_p) begin
                        if (code == lfsr[1:0]) lfsr  <= lfsr_next;  // correct -> next round
                        else                   state <= FINAL;      // wrong -> game over
                    end
                    sec_cnt <= '0; secs <= '0;
                end
                FINAL: begin
                    if (sec_cnt == ONE_SEC - 1) begin
                        sec_cnt <= '0;
                        if (secs == 2'd2) begin                // 3 seconds have passed
                            state <= INIT; lfsr <= 4'b0001; secs <= '0;
                        end else secs <= secs + 2'd1;
                    end else sec_cnt <= sec_cnt + 27'd1;
                end
                default: state <= INIT;
            endcase
        end
    end

    always_comb begin
        case (state)
            INIT:    LED = {9'b0, 2'b00,     1'b0, lfsr};
            PLAY:    LED = {9'b0, lfsr[1:0], 1'b0, lfsr};
            default: LED = 16'hFFFF;
        endcase
    end
endmodule

module btn_pulse #(parameter int DB = 16) (
    input logic clk, input logic rst, input logic btn, output logic pulse
);
    logic [DB-1:0] div;
    logic [3:0]    samples;
    logic          stable, stable_d;
    assign stable = &samples;
    always_ff @(posedge clk or posedge rst) begin
        if (rst) begin div <= '0; samples <= '0; stable_d <= 1'b0; end
        else begin
            div <= div + 1'b1;
            if (div == '0) samples <= {samples[2:0], btn};
            stable_d <= stable;
        end
    end
    assign pulse = stable & ~stable_d;
endmodule
重點
  • 每個按鈕都要經過 debounce + one-pulse，「按一下」才是一個事件。
  • LFSR 永遠不能是 0000（全 0 時 XOR 結果也是 0，會卡死）。
  • 「按錯」時 LFSR 不前進，所以前進的條件要寫在 code == lfsr[1:0] 的分支裡。

Lab 4 advanced 摘要（七段顯示貪食蛇）

  • PS/2 鍵盤（方向鍵、W/S 音量、Enter、Space）；蛇在 4 位七段顯示器的筆劃上移動。
  • 每 226 週期走一步：讀輸入 → 查路由表求下一個節點 → 判斷撞牆/咬自己 → 吃到肉就不 pop 尾巴。
  • 蛇身用 32 格 FIFO + head/tail 指標（見 記憶體）。
  • LFSR 每拍更新，產生候選的肉位置；落在蛇身上就等下一拍。

鍵盤與音效不在期中考常見範圍內，但 FIFO、LFSR、FSM、tick 都是。

Lab 5：Random Memory Access（新範圍）

8 格 × 4 bits 的 RAM，一個讀埠一個寫埠，write-first，每格有 valid bit，並統計有效格數。完整的圖解與波形在 設計模式 → 記憶體。

輸入情況下一拍
rst全部 memory、valid、cnt_valid、dout 清 0（同步）
wememory[wi] ← din、valid[wi] ← 1；原本 invalid 才 cnt_valid + 1
re 且 valid[ri]dout ← memory[ri]
re 且 invaliddout ← 0
we && re && wi == ridout ← din（write-first）
!redout ← 0
解答：lab5_practice.sv（助教 testbench 13 組 All Test Cases Passed）
// Lab 5 Practice — 8 x 4-bit RAM, 1 read + 1 write port, write-first, valid bits
`timescale 1ns/1ps
module lab5_practice (
    input  logic       clk,
    input  logic       rst,        // synchronous
    input  logic       we,
    input  logic [6:0] w_addr,
    input  logic [3:0] din,
    input  logic       re,
    input  logic [6:0] r_addr,
    output logic [3:0] dout,
    output logic [3:0] cnt_valid
);
    logic [3:0] memory [0:7];
    logic [7:0] valid;
    logic [2:0] wi, ri;            // mapped 3-bit indices

    // XOR folding: 7-bit address -> 3-bit index
    assign wi = w_addr[2:0] ^ w_addr[6:4] ^ w_addr[3:1];
    assign ri = r_addr[2:0] ^ r_addr[6:4] ^ r_addr[3:1];

    always_ff @(posedge clk) begin
        if (rst) begin
            for (int k = 0; k < 8; k++) memory[k] <= 4'd0;
            valid     <= 8'd0;
            cnt_valid <= 4'd0;
            dout      <= 4'd0;
        end else begin
            // ---- write port ----
            if (we) begin
                memory[wi] <= din;
                valid[wi]  <= 1'b1;
                if (!valid[wi]) cnt_valid <= cnt_valid + 4'd1;   // first write only
            end
            // ---- read port (registered) ----
            if (!re)                    dout <= 4'd0;
            else if (we && wi == ri)    dout <= din;             // write-first bypass
            else if (valid[ri])         dout <= memory[ri];
            else                        dout <= 4'd0;
        end
    end
endmodule
三個常見錯誤
  1. 用原始 7-bit 位址比較讀寫是否衝突（要比對應後的 3-bit index）。
  2. 忘了 write-first 旁路：非阻塞指定下，同一拍讀到的 memory[ri] 還是舊值。
  3. 覆寫已經 valid 的格子也讓 cnt_valid + 1。

考試可能的變形：改成 read-first（同一格同時讀寫回傳舊值）、加上「讀出後清除 valid」、做成 FIFO/stack、統計某種條件的格數。模擬考 B 題就是 stack 版本。
